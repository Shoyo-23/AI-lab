import pygame
import heapq

CELL = 40
COLS, ROWS = 16, 12
WIDTH, HEIGHT = COLS * CELL, ROWS * CELL + 50

WHITE = (245, 245, 245)
BLACK = (20, 20, 20)
BLUE = (66, 135, 245)
RED = (244, 67, 54)
ORANGE = (255, 152, 0)
PURPLE = (156, 39, 176)

walls = {(r, 6) for r in range(3, 9)}
walls |= {(3, c) for c in range(8, 13)}
walls |= {(r, 11) for r in range(5, 11)}

def distance(a, b):
    return abs(a[0] - b[0]) + abs(a[1] - b[1])

def inside(pos):
    r, c = pos
    return 0 <= r < ROWS and 0 <= c < COLS

def neighbors(pos):
    r, c = pos
    for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
        nxt = (r + dr, c + dc)
        if inside(nxt) and nxt not in walls:
            yield nxt

def a_star(start, goal):
    if start == goal:
        return []

    queue = [(distance(start, goal), 0, start)]
    parent = {}
    cost = {start: 0}

    while queue:
        _, g, current = heapq.heappop(queue)

        if current == goal:
            path = [current]
            while path[-1] != start:
                path.append(parent[path[-1]])
            return list(reversed(path))[1:]

        for nxt in neighbors(current):
            new_cost = g + 1
            if nxt not in cost or new_cost < cost[nxt]:
                cost[nxt] = new_cost
                parent[nxt] = current
                heapq.heappush(
                    queue,
                    (new_cost + distance(nxt, goal), new_cost, nxt)
                )

    return []

class EnemyBot:
    def __init__(self):
        self.pos = (1, 14)
        self.health = 100
        self.state = "PATROL"

    def decide_state(self, player):
        d = distance(self.pos, player)

        if self.health < 30:
            self.state = "FLEE"
        elif d <= 1:
            self.state = "ATTACK"
        elif d <= 6:
            self.state = "CHASE"
        else:
            self.state = "PATROL"

    def act(self, player):
        if self.state == "ATTACK":
            return 10

        if self.state == "CHASE":
            path = a_star(self.pos, player)

        elif self.state == "FLEE":
            cells = [
                (r, c)
                for r in range(ROWS)
                for c in range(COLS)
                if (r, c) not in walls
            ]
            target = max(cells, key=lambda p: distance(p, player))
            path = a_star(self.pos, target)

        else:
            target = (1, 14)
            path = a_star(self.pos, target)

        if path:
            self.pos = path[0]

        return 0

def draw_cell(screen, pos, color):
    r, c = pos
    rect = pygame.Rect(
        c * CELL + 5, r * CELL + 5,
        CELL - 10, CELL - 10
    )
    pygame.draw.rect(screen, color, rect, border_radius=5)

def main():
    pygame.init()

    screen = pygame.display.set_mode((WIDTH, HEIGHT))
    pygame.display.set_caption("AI Enemy Bot Game")
    clock = pygame.time.Clock()
    font = pygame.font.SysFont("consolas", 18)

    player = (10, 1)
    player_hp = 100
    enemy = EnemyBot()
    running = True

    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False

            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_ESCAPE:
                    running = False

                elif event.key in (
                    pygame.K_UP, pygame.K_DOWN,
                    pygame.K_LEFT, pygame.K_RIGHT
                ):
                    r, c = player

                    moves = {
                        pygame.K_UP: (r - 1, c),
                        pygame.K_DOWN: (r + 1, c),
                        pygame.K_LEFT: (r, c - 1),
                        pygame.K_RIGHT: (r, c + 1)
                    }

                    new_pos = moves[event.key]

                    if inside(new_pos) and new_pos not in walls:
                        player = new_pos

                    enemy.decide_state(player)
                    player_hp -= enemy.act(player)

                elif event.key == pygame.K_SPACE:
                    if distance(player, enemy.pos) <= 1:
                        enemy.health -= 15

        screen.fill(BLACK)

        for r in range(ROWS):
            for c in range(COLS):
                rect = pygame.Rect(c * CELL, r * CELL, CELL, CELL)
                color = (90, 90, 90) if (r, c) in walls else WHITE
                pygame.draw.rect(screen, color, rect)
                pygame.draw.rect(screen, (220, 220, 220), rect, 1)

        draw_cell(screen, player, BLUE)

        colors = {
            "PATROL": BLUE,
            "CHASE": ORANGE,
            "ATTACK": RED,
            "FLEE": PURPLE
        }

        draw_cell(screen, enemy.pos, colors[enemy.state])

        hud = (
            f"Player HP: {max(player_hp, 0)}  "
            f"Enemy HP: {max(enemy.health, 0)}  "
            f"State: {enemy.state}"
        )

        screen.blit(
            font.render(hud, True, WHITE),
            (8, ROWS * CELL + 12)
        )

        pygame.display.flip()
        clock.tick(8)

    pygame.quit()

if __name__ == "__main__":
    main()
